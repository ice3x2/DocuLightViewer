## 구현 체크리스트

- [ ] Phase 3-1: `viewer.js` — `collectNavFiles(node)` 함수 추가 `FR-26-003`
  - **대상 파일**: `src/renderer/viewer.js`
  - **삽입 위치**: `renderSidebarTree` 함수 근처 (L1572 전후)
  ```javascript
  /**
   * 사이드바 트리를 DFS 순회하여 모든 파일 노드를 순서대로 수집
   * 가상 노드(isVirtual, __external_links__)와 그 하위는 제외
   * @param {Object} node - 트리 노드
   * @returns {Array} 파일 노드 배열 (정렬 순서 유지)
   */
  function collectNavFiles(node) {
    if (!node) return [];
    if (node.isVirtual || node.title === '__external_links__') return [];
    if (!node.isDirectory) return [node];
    const result = [];
    if (node.children) {
      for (const child of node.children) {
        result.push(...collectNavFiles(child));
      }
    }
    return result;
  }
  ```

- [ ] Phase 3-2: `viewer.js` — `updateDocNavBox()` 함수 추가 `FR-26-003`
  - **대상 파일**: `src/renderer/viewer.js`
  - **삽입 위치**: `collectNavFiles` 함수 바로 아래
  ```javascript
  /**
   * 뷰어 하단 이전/다음 파일 박스 업데이트
   * NFR-26-001: 기존 박스 재사용 (완전 재생성 금지)
   */
  function updateDocNavBox() {
    const state = window.DocuLight && window.DocuLight.state;
    if (!state || !state.sidebarTree || !state.currentFilePath) {
      hideDocNavBox();
      return;
    }

    const files = collectNavFiles(state.sidebarTree);
    const idx = files.findIndex(f => f.path === state.currentFilePath);
    if (idx < 0) {
      hideDocNavBox();
      return;
    }

    const prev = idx > 0 ? files[idx - 1] : null;
    const next = idx < files.length - 1 ? files[idx + 1] : null;

    if (!prev && !next) {
      hideDocNavBox();
      return;
    }

    ensureDocNavBox();
    const prevEl = document.getElementById('doc-nav-prev');
    const nextEl = document.getElementById('doc-nav-next');

    updateNavItem(prevEl, prev, 'prev');
    updateNavItem(nextEl, next, 'next');
  }

  function hideDocNavBox() {
    const box = document.getElementById('doc-nav-box');
    if (box) box.style.display = 'none';
  }

  function updateNavItem(el, node, direction) {
    if (!el) return;
    if (!node) {
      el.style.display = 'none';
      el.removeAttribute('data-path');
      return;
    }
    el.style.display = '';
    const name = getDisplayName(node);
    const titleEl = el.querySelector('.doc-nav-title');
    if (titleEl) titleEl.textContent = name;
    const label = direction === 'prev'
      ? (t('docNav.prevAriaLabel', { name }) || `이전: ${name}`)
      : (t('docNav.nextAriaLabel', { name }) || `다음: ${name}`);
    el.setAttribute('aria-label', label);
    el.dataset.path = node.path;
  }
  ```

- [ ] Phase 3-3: `viewer.js` — `ensureDocNavBox()` 함수 (최초 DOM 생성) `IR-26-002`
  - **대상 파일**: `src/renderer/viewer.js`
  - **삽입 위치**: `updateDocNavBox` 함수 바로 아래
  ```javascript
  /**
   * #doc-nav-box DOM 요소가 없으면 생성하고, 있으면 재사용 (display 복원)
   * #content 다음 형제로 삽입
   */
  function ensureDocNavBox() {
    let box = document.getElementById('doc-nav-box');
    if (box) {
      box.style.display = '';
      return;
    }
    const contentEl = document.getElementById('content');
    if (!contentEl) return;

    box = document.createElement('nav');
    box.id = 'doc-nav-box';
    box.setAttribute('role', 'navigation');
    box.setAttribute('aria-label', t('docNav.ariaLabel') || '문서 네비게이션');

    const prevEl = document.createElement('a');
    prevEl.id = 'doc-nav-prev';
    prevEl.className = 'doc-nav-item doc-nav-prev';
    prevEl.setAttribute('tabindex', '0');
    prevEl.innerHTML = `<span class="doc-nav-arrow">←</span>
      <span class="doc-nav-label">${t('docNav.prev') || '이전'}</span>
      <span class="doc-nav-title"></span>`;

    const nextEl = document.createElement('a');
    nextEl.id = 'doc-nav-next';
    nextEl.className = 'doc-nav-item doc-nav-next';
    nextEl.setAttribute('tabindex', '0');
    nextEl.innerHTML = `<span class="doc-nav-title"></span>
      <span class="doc-nav-label">${t('docNav.next') || '다음'}</span>
      <span class="doc-nav-arrow">→</span>`;

    box.appendChild(prevEl);
    box.appendChild(nextEl);
    contentEl.insertAdjacentElement('afterend', box);

    // 클릭/키보드 이벤트 (한 번만 등록)
    [prevEl, nextEl].forEach(el => {
      el.addEventListener('click', () => {
        const targetPath = el.dataset.path;
        if (!targetPath) return;
        // 탭 모드 처리 (viewer.js L1652-1659 패턴 재사용)
        if (window.DocuLight && window.DocuLight.fn && window.DocuLight.fn.navigateToForTab) {
          window.DocuLight.fn.navigateToForTab(targetPath);
        } else {
          window.doclight.navigateTo(targetPath);
        }
      });
      el.addEventListener('keydown', (e) => {
        if (e.key === 'Enter' || e.key === ' ') {
          e.preventDefault();
          el.click();
        }
      });
    });
  }
  ```

## 언어+버전 주의사항
- `insertAdjacentElement('afterend', el)` — Electron Chromium 지원, `#content`가 반드시 DOM에 존재해야 함
- `ensureDocNavBox()`: 이벤트 리스너는 최초 DOM 생성 시 **한 번만** 등록. 함수가 여러 번 호출되어도 리스너 중복 등록 없음 (박스 재사용으로 보장)
- `el.dataset.path` — 특수문자 포함 경로 저장 시 문제없음

---

## LLM 코딩 주의사항
- **Simplicity**: `collectNavFiles()` 재귀 DFS는 10줄 이내. visited set 불필요 (트리 DAG 보장됨). 복잡한 최적화 금지.
- **Surgical**: `onRenderMarkdown` 핸들러 끝에 `updateDocNavBox();` 한 줄만 추가. 기존 핸들러 로직 수정 없음. `onSidebarTree` 동일.
- **탭 모드**: 클릭 이벤트에서 반드시 `navigateToForTab` 우선 → `navigateTo` fallback 패턴 사용 (SRS 제약사항 4조).
- **정렬 불변 (제약사항 1조)**: `collectNavFiles()`가 DFS로 트리를 순회하면 기존 트리 표시 순서(파일명 기준 정렬)와 일치함. 별도 정렬 로직 추가 불필요, 추가하지 않는다.

## 테스트 시나리오

### 정상 케이스
- Given: 디렉토리에 A.md, B.md, C.md 있고 B.md 열림, 사이드바 트리 존재
  When: 뷰어 하단 확인
  Then: 좌측에 "← A.md" (또는 frontmatter name), 우측에 "C.md →" 표시. 클릭 시 해당 파일 열림.

### 예외 케이스
- Given: 사이드바 트리 없음 (직접 파일 드래그앤드롭으로 열기)
  When: 뷰어 하단 확인
  Then: `#doc-nav-box` 미표시 (display:none 또는 DOM 미존재)
- Given: 현재 파일이 사이드바 트리 파일 목록에 없는 경우
  When: 뷰어 하단 확인
  Then: `#doc-nav-box` 미표시

### 경계값 테스트
- Given: 파일 목록에서 첫 번째 파일 열림
  When: 하단 박스 확인
  Then: 좌측(이전) 박스 display:none, 우측(다음) 박스만 표시
- Given: 파일 목록에서 마지막 파일 열림
  When: 하단 박스 확인
  Then: 우측(다음) 박스 display:none, 좌측(이전) 박스만 표시
- Given: `__external_links__` 가상 노드 하위 파일이 현재 열린 경우
  When: 하단 박스 확인
  Then: `#doc-nav-box` 미표시 (가상 노드 하위는 collectNavFiles에서 제외됨)
